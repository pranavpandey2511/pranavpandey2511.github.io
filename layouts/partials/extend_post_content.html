{{- /* Previous/next part navigation for posts that belong to a series. */}} {{- $terms := .GetTerms "series" }} {{- with $terms }} {{- $term := index . 0 }} {{- $pages := sort (sort $term.Pages "Date" "asc") "Params.series_order" "asc" }} {{- $prev := "" }}{{- $next := "" }} {{- range $i, $p := $pages }} {{- if eq $p.Permalink $.Permalink }} {{- if gt $i 0 }}{{ $prev = index $pages (sub $i 1) }}{{ end }} {{- if lt (add $i 1) (len $pages) }}{{ $next = index $pages (add $i 1) }}{{ end }} {{- end }} {{- end }} {{- if or $prev $next }}
{{- with $prev }} « Previous part
{{ .Title }} {{- end }} {{- with $next }} Next part »
{{ .Title }} {{- end }}
{{- end }} {{- end }}
